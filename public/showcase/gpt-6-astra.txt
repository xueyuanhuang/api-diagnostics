<script>
  const scene = document.getElementById("scene");
  const toggle = document.getElementById("toggle");
  const motionPreference = window.matchMedia("(prefers-reduced-motion: reduce)");
  let paused = false;

  function setPaused(value) {
    paused = value;

    if (paused) {
      scene.pauseAnimations();
    } else {
      scene.unpauseAnimations();
    }

    toggle.textContent = paused ? "▶ 播放动画" : "Ⅱ 暂停动画";
    toggle.setAttribute("aria-label", paused ? "播放动画" : "暂停动画");
  }

  toggle.addEventListener("click", () => setPaused(!paused));
  motionPreference.addEventListener("change", event => setPaused(event.matches));

  setPaused(motionPreference.matches);
</script>
